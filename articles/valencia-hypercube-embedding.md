coordinate embedding implemented by
[`linf.hypercube.embedding()`](https://pgajer.github.io/linf/reference/linf.hypercube.embedding.md)
on a bundled VALENCIA-derived example dataset. The source object,
`valencia_linf_hypercube_1k`, is generated by
`data-raw/build_valencia_linf_hypercube_1k.R` from the VALENCIA 13k
training set. It keeps four phylotype coordinates,

## Embedding Helpers

The figures below use two interactive three-dimensional views.

1.  Classical MDS embeds the Euclidean distance matrix into three
    dimensions.
2.  If the optional `grip` package is available, edge-KK starts from the
    MDS coordinates and refines the layout using a local
    k-nearest-neighbor graph whose edge lengths are measured in the
    source coordinate system.

``` r

if (!requireNamespace("plotly", quietly = TRUE)) {
  stop("This article preview requires the plotly package.")
}
if (!requireNamespace("htmltools", quietly = TRUE)) {
  stop("This article preview requires the htmltools package.")
}

has_grip <- requireNamespace("grip", quietly = TRUE)
edge_kk_fun <- NULL
if (has_grip && exists("grip.optimize.edge.kk.layout", envir = asNamespace("grip"))) {
  edge_kk_fun <- getFromNamespace("grip.optimize.edge.kk.layout", "grip")
}

mds3 <- function(X) {
  d <- stats::dist(X)
  fit <- stats::cmdscale(d, k = 3, eig = TRUE)
  coords <- as.matrix(fit$points)
  colnames(coords) <- c("Dim1", "Dim2", "Dim3")
  coords
}

knn_edges <- function(X, k = 8L) {
  D <- as.matrix(stats::dist(X))
  n <- nrow(D)
  edges <- vector("list", n)
  weights <- vector("list", n)

  for (i in seq_len(n)) {
    ord <- order(D[i, ], decreasing = FALSE)
    nbrs <- ord[ord != i][seq_len(min(k, n - 1L))]
    keep <- nbrs > i
    if (any(keep)) {
      nbrs <- nbrs[keep]
      edges[[i]] <- cbind(i, nbrs)
      weights[[i]] <- D[cbind(i, nbrs)]
    }
  }

  edge_matrix <- do.call(rbind, edges[lengths(edges) > 0L])
  edge_weights <- unlist(weights[lengths(weights) > 0L], use.names = FALSE)

  list(edges = edge_matrix, weights = pmax(edge_weights, .Machine$double.eps))
}

edge_kk3 <- function(X, initial, k = 8L, max_iter = 20L) {
  if (is.null(edge_kk_fun)) {
    return(NULL)
  }

  graph <- knn_edges(X, k = k)
  out <- edge_kk_fun(
    coords = initial,
    edges = graph$edges,
    n = nrow(X),
    edge_weights = graph$weights,
    dim = 3L,
    max_iter = max_iter,
    return_trace = FALSE,
    diagnostics = FALSE,
    seed = 1L
  )

  coords <- as.matrix(out$coords)
  colnames(coords) <- c("Dim1", "Dim2", "Dim3")
  coords
}

plot_embedding <- function(coords, title, subtitle) {
  df <- data.frame(
    coords,
    dominant_component = meta$dominant_component,
    Val_CST = meta$Val_CST,
    selected_mass = meta$selected_mass,
    sample_id = meta$sample_id,
    stringsAsFactors = FALSE
  )

  hover <- paste0(
    "sample: ", df$sample_id,
    "<br>dominant: ", df$dominant_component,
    "<br>Val_CST: ", df$Val_CST,
    "<br>selected mass: ", signif(df$selected_mass, 4)
  )

  p <- plotly::plot_ly(
    df,
    x = ~Dim1,
    y = ~Dim2,
    z = ~Dim3,
    color = ~dominant_component,
    symbol = ~Val_CST,
    text = hover,
    hoverinfo = "text",
    type = "scatter3d",
    mode = "markers",
    marker = list(size = 3, opacity = 0.75)
  )

  plotly::layout(
    p,
    title = list(text = paste0(title, "<br><sup>", subtitle, "</sup>")),
    scene = list(
      xaxis = list(title = "Dim 1"),
      yaxis = list(title = "Dim 2"),
      zaxis = list(title = "Dim 3")
    ),
    legend = list(orientation = "h")
  )
}

embedding_panel <- function(label, X, reference = NULL) {
  title <- if (is.null(reference)) {
    paste0(label, ": 4D composition")
  } else {
    paste0(label, ": reference ", reference)
  }

  mds_coords <- mds3(X)
  widgets <- list(
    htmltools::tags$h3(paste0(title, " -- classical MDS")),
    plot_embedding(
      mds_coords,
      title = paste0(title, " -- MDS"),
      subtitle = "Classical MDS of Euclidean distances in the source coordinates."
    )
  )

  kk_coords <- edge_kk3(X, mds_coords)
  if (!is.null(kk_coords)) {
    widgets <- c(
      widgets,
      list(
        htmltools::tags$h3(paste0(title, " -- edge-KK")),
        plot_embedding(
          kk_coords,
          title = paste0(title, " -- edge-KK"),
          subtitle = "MDS initialization refined on an 8-nearest-neighbor graph with source-space edge lengths."
        )
      )
    )
  } else {
    widgets <- c(
      widgets,
      list(htmltools::tags$p("Optional edge-KK layout skipped because the grip package is not available."))
    )
  }

  htmltools::tagList(widgets)
}
```

## Reproducibility Notes

The bundled dataset is intentionally small enough to ship with `linf`.
It is created from the VALENCIA 13k source matrix by:

1.  selecting `Li`, `Lc`, `Gv`, and `Bv`;
2.  dropping rows with zero total mass in these four selected
    components;
3.  L1-normalizing over the selected components;
4.  stratifying by the dominant selected component; and
5.  sampling 1,000 rows with `set.seed(20261604)`.

The four hypercube charts are generated at render time using
[`linf.hypercube.embedding()`](https://pgajer.github.io/linf/reference/linf.hypercube.embedding.md),
so this article previews both the bundled data and the exported
embedding API.
